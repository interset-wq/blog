## utils/

### db.py

```python
import json
from pathlib import Path
from uuid import UUID
DB_PATH = Path(__file__).parent.parent / "data/users.json"

def read_users() -> list[dict]:
    with open(DB_PATH, encoding="utf-8") as f:
        return json.load(f)

def write_users(updated: list[dict]) -> None:
    with open(DB_PATH, "w", encoding="utf-8") as f:
        json.dump(updated, f, indent=2, ensure_ascii=False)

def read_by_email(email: str) -> dict | None:
    for user in read_users():
        if user["email"] == email:
            return user
    return None

def read_by_id(id: UUID) -> dict | None:
    for user in read_users():
        if user["id"] == str(id):
            return user
    return None
```

### jwt.py

封装 PyJWT，用于签发和验证 JWT。

```python
from datetime import datetime, timedelta, UTC
import os
import jwt
from jwt.exceptions import InvalidTokenError
from dotenv import load_dotenv

load_dotenv(override=True)
jwt_secret = os.environ.get("JWT_SECRET")
algo = "HS256"

def sign_token(payload: dict) -> str:
    """签发JWT，附带过期时间"""
    new_payload = payload.copy()
    new_payload["exp"] = datetime.now(tz=UTC) + timedelta(days=1)
    return jwt.encode(new_payload, key=jwt_secret, algorithm=algo)

def verify_token(token: str) -> dict | None:
    """校验JWT，过期/无效返回None"""
    try:
        return jwt.decode(token, jwt_secret, algorithms=[algo],
                          options={"verify_exp": True})
    except InvalidTokenError:
        return None
```

| 差异点 | Express（jsonwebtoken） | FastAPI（PyJWT） |
| --- | --- | --- |
| 过期时间 | 选项里写 `expiresIn: "1d"`，库自动塞 `exp` | 手动把 `exp`放进 payload |
| 解码算法 | `jwt.verify` 默认认可常见算法 | 必须显式传 `algorithms=["HS256"]` 白名单，防止算法混淆攻击 |

### auth.py

- POST `/register`：注册成功立即签发 JWT（注册即登录）
- POST `/login`：登录成功签发 JWT
- POST `/logout`：需要鉴权，把当前 token 拉黑
- GET `/profile`：需要鉴权，返回解码后的 payload
- POST `/token`：表单格式登录端点，专供 Swagger 文档的 Authorize 弹窗使用（OAuth2 密码流发的是表单不是 JSON），前端正常走 `/login` 即可

```python
from typing import Annotated
from uuid import uuid4
from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordRequestForm
from pwdlib import PasswordHash
from pydantic import BaseModel
from pwdlib.hashers.bcrypt import BcryptHasher
from deps import CurrentUser, Token
from utils.db import read_by_email, read_users, write_users
from utils.jwt import sign_token
from utils.token_blacklist import blacklist_token

password_hash = PasswordHash((BcryptHasher(),))
router = APIRouter(prefix="/auth", tags=["auth"])

class Credentials(BaseModel):
    """JSON 请求体模型：字段缺失/类型不符时，FastAPI 自动拦截并返回 422"""
    email: str
    password: str

@router.post("/register", status_code=status.HTTP_201_CREATED)
def register(body: Credentials) -> dict:
    if not body.email or not body.password:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Email and password are required",
        )
    if read_by_email(body.email):
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Email already in use",
        )
    new_user = {
        "id": str(uuid4()),
        "email": body.email,
        "passwordHash": password_hash.hash(body.password),
        "provider": "local",
        "role": "user",
    }
    users = read_users()
    users.append(new_user)
    write_users(users)

    # payload 只放身份信息（不放密码），客户端每次请求带上它，
    # 服务端解码即可知道"是谁、什么角色"，无需查库
    token = sign_token(
        {"id": new_user["id"], "email": new_user["email"], "role": new_user["role"]}
    )
    return {"message": "User registered successfully", "token": token}

@router.post("/login")
def login(body: Credentials) -> dict:
    user = read_by_email(body.email)
    if not user or not password_hash.verify(body.password, user["passwordHash"]):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid credentials",
        )
    token = sign_token(
        {"id": user["id"], "email": user["email"], "role": user["role"]}
    )
    return {"message": "Login successful", "token": token}

@router.post("/logout")
def logout(token: Token, user: CurrentUser) -> dict:
    # 两个依赖参数分工：CurrentUser 保证"必须先登录"（无效 token 到不了这里），
    # Token 拿原始 token 字符串去拉黑
    blacklist_token(token)
    return {"message": "Logged out successfully"}

@router.get("/profile")
def profile(user: CurrentUser) -> dict:
    return {"user": user}

@router.post("/token")
def token_for_docs(form: Annotated[OAuth2PasswordRequestForm, Depends()]) -> dict:
    """表单格式登录端点，供 Swagger Authorize 弹窗使用（OAuth2 密码流）"""
    user = read_by_email(form.username)
    if not user or not password_hash.verify(form.password, user["passwordHash"]):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid credentials",
            headers={"WWW-Authenticate": "Bearer"},
        )
    access_token = sign_token(
        {"id": user["id"], "email": user["email"], "role": user["role"]}
    )
    # OAuth2 密码流规定返回 access_token / token_type 字段
    return {"access_token": access_token, "token_type": "bearer"}
```

注意参数标注 `body: Credentials`：请求体的解析和校验由 Pydantic 模型声明完成，函数内拿到的 `body` 已保证字段齐全、类型正确——Express 里 `req.body.email` 手动取值 + 手动判空的活，这里在参数表上就完成了（细节见下一篇"类型注解的魔法"）。

### admin.py

- GET `/users`：需要鉴权和授权，仅 admin 可用

```python
from typing import Annotated
from fastapi import APIRouter, Depends
from deps import require_role
from utils.db import read_users

router = APIRouter(prefix="/admin", tags=["admin"])

@router.get("/users")
def list_users(user: Annotated[dict, Depends(require_role("admin"))]) -> dict:
    users = [
        # 每个user排除passwordHash字段
        {k: v for k, v in u.items() if k != "passwordHash"} for u in read_users()
    ]
    return {"users": users}
```

`require_role("admin")` 是个工厂函数：调用后返回一个现成的依赖，对应 Express 的 `authorizeRole("admin")` 中间件。

## deps.py：鉴权与授权（对标 middleware/）

Express 版在这里写两个中间件函数；FastAPI 把同样的逻辑写成**依赖**，声明在路由参数上。

### 提取 token：OAuth2PasswordBearer

```python
from fastapi.security import OAuth2PasswordBearer
reusable_oauth2 = OAuth2PasswordBearer(tokenUrl="/api/auth/token")
```

- 它只负责从 `Authorization: Bearer <token>` 提取 token，**本身不做任何校验**，校验仍由黑名单 + 验签完成
- token 缺失时自动返回 401（带 `WWW-Authenticate: Bearer` 头）——Express 版要手写 `authHeader.startsWith("Bearer ")` 判断
- `tokenUrl` 是给 `/docs` 的 Swagger 文档看的元信息（Authorize 按钮），指向签发 token 的接口，即上面 auth.py 里的 `/token` 端点

### 鉴权：get_current_user

```python
def get_current_user(token: Annotated[str, Depends(reusable_oauth2)]) -> dict:
    """token 有效则返回解码后的 payload（含 id/email/role/exp）"""
    if is_blacklisted(token):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Token has been invalidated. Log in again.",
        )
    decoded = verify_token(token)
    if not decoded:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token",
        )
    return decoded
```

**依赖可以嵌套**：参数里的 `Depends(reusable_oauth2)` 会先执行拿到原始 token 字符串，这里再依次做黑名单检查、验签——Express 中间件靠 `next()` 串起来的链条，在 FastAPI 里是依赖函数的参数声明。

### 授权：require_role 工厂

```python
def require_role(role: str):
    """授权依赖的工厂：角色在写路由时才确定，所以返回一个现成的依赖"""
    def _authorize(user: Annotated[dict, Depends(get_current_user)]) -> dict:
        """先登录，再授权"""
        if not user or user.get("role") != role:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Access denied",
            )
        return user
    return _authorize
```

### 常用依赖的短别名

```python
# get_current_user 依赖 reusable_oauth2
# reusable_oauth2（提取token） -> get_current_user（黑名单->解码token）
# 走完整鉴权链（提取→黑名单→验签），拿解码后的用户信息
CurrentUser = Annotated[dict, Depends(get_current_user)]
# 只提取不校验的原始 token 字符串（logout 加黑名单用）
Token = Annotated[str, Depends(reusable_oauth2)]
```

路由参数直接写 `user: CurrentUser`，比完整的 `Annotated[dict, Depends(get_current_user)]` 短得多——`Annotated` 类型别名让依赖声明可以复用。
