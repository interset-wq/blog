## `<stdio.h>` — 输入输出

最常用的库，负责标准输入输出与文件操作。

```c
#include <stdio.h>

int main(void) {
    // --- 格式化输出到控制台 ---
    int n = 42;
    printf("int: %d, hex: %x, octal: %o\n", n, n, n);  // 十进制/十六进制/八进制
    printf("float: %.2f, char: %c, string: %s\n", 3.14, 'A', "hello");

    // --- 格式化输出到字符串（sprintf / snprintf）---
    char buf[100];
    snprintf(buf, sizeof(buf), "value=%d", n);  // 写进字符串（n 版本更安全，防溢出）
    puts(buf);                                  // 输出一行 + 换行

    // --- 从字符串读入（sscanf，反方向）---
    int a, b;
    sscanf("12 34", "%d %d", &a, &b);           // 从字符串按格式解析 → a=12, b=34

    // --- 格式化输入 ---
    // scanf("%d", &n);              // 从键盘读取（注意 &）

    // --- 文件读写（文本 + 二进制）---
    FILE *fp = fopen("out.txt", "w");
    if (fp) {
        fprintf(fp, "Hello %d\n", n);           // 写入文件
        fputs("another line\n", fp);            // 写字符串（自动带上 \n）
        fclose(fp);
    }

    FILE *in = fopen("out.txt", "r");
    char line[256];
    while (fgets(line, sizeof(line), in)) {     // 逐行读取（fgets 最安全）
        printf("read: %s", line);
    }
    fclose(in);
    return 0;
}
```

| 函数 | 作用 |
| --- | --- |
| `printf` / `fprintf` / `sprintf` | 格式化输出到 stdout / 文件 / 字符串 |
| `snprintf` | 写字符串（带长度限制，更安全） |
| `scanf` / `fscanf` / `sscanf` | 从 键盘 / 文件 / 字符串 格式化读入 |
| `puts` / `fputs` | 输出字符串（`puts` 自带换行） |
| `fopen` / `fclose` | 打开 / 关闭文件 |
| `fgets` | 逐行读取（安全，防缓冲区溢出） |
| `getchar` / `putchar` / `fgetc` / `fputc` | 单字符读 / 写 |
| `fread` / `fwrite` | 二进制块读写 |

> `scanf` 取地址需加 `&`（数组名本身是指针除外）；字符串读入建议用 `fgets` 避免缓冲区溢出。

### 二进制读写（`fread` / `fwrite`）

```c
#include <stdio.h>

struct Point { int x, y; };

int main(void) {
    struct Point p = {3, 4};

    // 整个结构体一次性写入 / 读回
    FILE *fp = fopen("point.bin", "wb");
    fwrite(&p, sizeof(struct Point), 1, fp);   // 写 1 块 sizeof(Point) 字节
    fclose(fp);

    FILE *in = fopen("point.bin", "rb");
    struct Point q;
    fread(&q, sizeof(struct Point), 1, in);    // 读回
    fclose(in);
    printf("(%d,%d)\n", q.x, q.y);             // (3,4)
    return 0;
}
```

### 单字符 I/O

```c
#include <stdio.h>

int main(void) {
```

## `<string.h>` — 字符串与内存

C 没有内置字符串，靠 `char[]` + `\0`，字符串函数都在这里。

### 字符串操作：拷贝 / 拼接 / 长度 / 比较

```c
#include <stdio.h>
#include <string.h>

int main(void) {
    char a[20] = "Hello";
    char b[20];

    // 拷贝
    strcpy(b, a);             // 把 a 拷贝进 b
    strncpy(b, "Hi", 2);      // 限长拷贝（只拷贝前 2 个字符，更安全）

    // 拼接
    strcat(a, " World");      // a 后面拼上，a 变成 "Hello World"
    strncat(a, "!", 1);       // 限长拼接

    // 长度（不含 \0）
    size_t len = strlen(a);
    printf("len=%zu\n", len);

    // 比较：返回 <0 / 0 / >0（按字典序）
    printf("%d\n", strcmp("abc", "abd"));    // 负值（c < d）
    printf("%d\n", strcmp("abc", "abc"));    // 0（相等）
    printf("%d\n", strncmp("abc", "abd", 2)); // 0（前 2 个字符相等）

    return 0;
}
```

> **`n` 版本（`strncpy`/`strncat`/`strncmp`）更安全**：限制了最多处理多少字符，避免越界。但 `strncpy` 若源串短于 n 会用 `\0` 填满，需注意。

### 查找：字符 / 子串

```c
#include <stdio.h>
#include <string.h>

int main(void) {
    const char *s = "hello world";

    char *p = strchr(s, 'o');       // 找第一个 'o' 的位置
    if (p) printf("at offset %ld\n", p - s);   // 4

    char *sub = strstr(s, "world"); // 找子串
    if (sub) printf("found: %s\n", sub);       // world

    // strrchr：从右往左找最后一个
    char *last = strrchr(s, 'o');   // 指向最后一个 'o'（offset 7）
    return 0;
}
```

### 内存操作：`memset` / `memcpy` / `memmove`

```c
#include <stdio.h>
#include <string.h>

int main(void) {
    char buf[10];

    memset(buf, 0, sizeof(buf));     // 全部置 0（常用：清零数组/结构体）
    memset(buf, 'A', 5);             // 前 5 字节填 'A'

    char src[10] = "hello";
    char dst[10];
    memcpy(dst, src, 5);             // 拷贝 5 字节（src 和 dst 不能重叠）

    // memmove：即使源和目的重叠也安全（memcpy 重叠时未定义行为）
    char m[10] = "hello";
    memmove(m + 1, m, 5);            // 自己搬自己，安全
    return 0;
}
```

> **区别**：`memcpy` 拷贝源/目的**不能重叠**；`memmove` 允许重叠，代价是稍慢。用字符串相关（拷贝/拼接）优先用 `str*` 函数，处理原始字节用 `mem*`。

### 分割字符串：`strtok`（常用）

```c
#include <stdio.h>
#include <string.h>

int main(void) {
    char line[] = "apple,banana,cherry";
    const char *delim = ",";

    char *tok = strtok(line, delim);    // 首次调用：传字符串
    while (tok) {
        printf("%s\n", tok);            // 依次输出 apple / banana / cherry
        tok = strtok(NULL, delim);      // 后续调用：传 NULL 继续分割
    }
    return 0;
}
```

> ⚠️ `strtok` 会**修改原字符串**（把分隔符替换成 `\0`），且是**线程不安全**的（内部有静态指针）。现代代码可用 `strtok_r`（可重入版）或在 C++ 用 `std::stringstream`。

| 函数 | 作用 |
| --- | --- |
| `strlen` | 长度（不含 `\0`） |
| `strcpy` / `strncpy` | 拷贝（`n` 版本限长更安全） |
| `strcat` / `strncat` | 拼接 |
| `strcmp` / `strncmp` | 比较（返回 0 表示相等） |
| `strchr` / `strrchr` / `strstr` | 找字符 / 找字符（反向）/ 找子串 |
| `strtok` / `strtok_r` | 按分隔符分割 |
| `memset` / `memcpy` / `memmove` | 内存置零 / 拷贝（不可重叠）/ 拷贝（可重叠） |

## `<ctype.h>` — 字符处理

判断字符类型与大小写转换。

```c
#include <stdio.h>
#include <ctype.h>

int main(void) {
    printf("%d\n", isdigit('5'));   // 1（真）数字
    printf("%d\n", isalpha('a'));   // 1 字母
    printf("%d\n", isalnum('5'));   // 1 字母或数字
    printf("%d\n", isspace(' '));   // 1 空白（空格/换行/制表）
    printf("%d\n", ispunct('!'));   // 1 标点
    printf("%d\n", islower('a'));   // 1 小写
    printf("%d\n", isupper('A'));   // 1 大写
    printf("%c\n", toupper('a'));   // A 转大写
    printf("%c\n", tolower('B'));   // b 转小写
    return 0;
}
```

> 所有 `is*` 函数在条件为真时返回**非 0**（不一定是 1），判断时写成 `if (isdigit(c))` 即可。注意这些函数只对 `char`（当作 `unsigned char` 或 `EOF`）安全，不要传负数。

| 函数 | 作用 |
| --- | --- |
| `isdigit` / `isalpha` / `isalnum` | 数字 / 字母 / 字母数字 |
| `isspace` / `ispunct` / `islower` / `isupper` | 空白 / 标点 / 小写 / 大写 |
| `toupper` / `tolower` | 转大写 / 小写 |

### 实用场景：统计文本里各类字符

```c
#include <stdio.h>
#include <ctype.h>

int main(void) {
    const char *s = "Hello 123, world!";
    int digits = 0, alpha = 0, punct = 0;

    for (int i = 0; s[i] != '\0'; i++) {
        if (isdigit(s[i])) digits++;
        else if (isalpha(s[i])) alpha++;
        else if (ispunct(s[i])) punct++;
    }
    printf("alpha=%d digits=%d punct=%d\n", alpha, digits, punct);
    // alpha=10 digits=3 punct=2
    return 0;
}
```

## 综合示例：统计字符串中数字个数

```c
#include <stdio.h>
#include <ctype.h>
#include <string.h>

int main(void) {
    char text[] = "abc123 x 42y zz";
    int digits = 0;
    for (size_t i = 0; i < strlen(text); i++) {
        if (isdigit(text[i])) digits++;
    }
    printf("digits: %d\n", digits);   // 5（1,2,3,4,2）
    return 0;
}
```
