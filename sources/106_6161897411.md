
`http` 是 Node 内置的 HTTP 模块，用来**直接**搭建 HTTP 服务器、发起 HTTP 请求。Express 等框架正是在它之上封装的。

## 创建一个 HTTP 服务器

`http.createServer(handler)` 创建服务器，`server.listen(port)` 开始监听。

```js
const http = require("http");

const server = http.createServer((req, res) => {
  res.writeHead(200, { "Content-Type": "text/plain; charset=utf-8" });
  res.end("Hello World!");
});

server.listen(3000, () => {
  console.log("Server running at http://localhost:3000");
});
```

### `req`（请求对象）

- `req.url`：请求路径与查询字符串。
- `req.method`：HTTP 方法（`GET`/`POST`/...）。
- `req.headers`：请求头对象。
- `req` 是一个 **Readable 流**，可通过事件读取请求体。

### `res`（响应对象）

- `res.writeHead(statusCode, headers)`：设置状态码和响应头。
- `res.write(data)`：写入响应体。
- `res.end(data?)`：结束响应（可带最后一段数据）。
- `res` 是一个 **Writable 流**。

## 基础路由

根据 `req.url` 和 `req.method` 分发不同处理逻辑：

```js
const http = require("http");

const server = http.createServer((req, res) => {
  const { url, method } = req;

  if (url === "/" && method === "GET") {
    res.writeHead(200, { "Content-Type": "text/html" });
    res.end("<h1>Home</h1>");
  } else if (url === "/api/users" && method === "GET") {
    res.writeHead(200, { "Content-Type": "application/json" });
    res.end(JSON.stringify([{ id: 1, name: "Alice" }]));
  } else if (url === "/api/users" && method === "POST") {
    // 读取请求体（req 是 Readable 流）
    let body = "";
    req.on("data", (chunk) => (body += chunk));
    req.on("end", () => {
      console.log("收到数据:", body);
      res.writeHead(201, { "Content-Type": "application/json" });
      res.end(JSON.stringify({ created: true }));
    });
  } else {
    res.writeHead(404, { "Content-Type": "text/plain" });
    res.end("Not Found");
  }
});

server.listen(3000);
```

## 发起 HTTP 请求（客户端）

`http.request(options, callback)` 向外部服务器发请求：

```js
const http = require("http");

const req = http.request(
  {
    hostname: "example.com",
    port: 80,
    path: "/api/data",
    method: "GET",
    headers: { "Content-Type": "application/json" },
  },
  (res) => {
    let data = "";
    res.on("data", (chunk) => (data += chunk)); // res 是可读流
    res.on("end", () => console.log("响应:", data));
  }
);

req.on("error", (err) => console.error("请求失败:", err));
req.end();
```

> 现代写法可用 `http.get(url, cb)` 快速发起 GET，或用 `fetch()`（Node 18+ 内置）替代。

## 发起 HTTP 请求：`fetch()`

Node 18 起内置了与浏览器一致的 **`fetch` API**，无需引入 `http` 模块即可发起请求，返回 Promise，是现代推荐的客户端写法。

### 基础 GET 请求

```js
const res = await fetch("https://api.example.com/users");
console.log(res.status);            // 200
console.log(res.ok);                // true（status 在 200-299 之间）

const data = await res.json();      // 解析 JSON 响应体
console.log(data);
```

### 请求配置对象

`fetch(url, options)` 的 `options` 支持 `method`、`headers`、`body` 等：

```js
// POST 提交 JSON
const res = await fetch("https://api.example.com/users", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ name: "Alice", age: 30 }),
});

if (!res.ok) {
  throw new Error(`请求失败: ${res.status}`);
}
const created = await res.json();
```

### 常用场景

```js
// 1. 读取纯文本
const text = await (await fetch(url)).text();

// 2. 携带查询参数（手动拼接或 URLSearchParams）
const params = new URLSearchParams({ q: "node", page: "2" });
const res = await fetch(`https://api.example.com/search?${params}`);

// 3. 处理错误（网络层失败会 reject，HTTP 错误状态需自行判断 res.ok）
try {
  const res = await fetch(url, { headers: { Authorization: `Bearer ${token}` } });
  if (res.status === 401) return console.log("未授权");
  const data = await res.json();
} catch (err) {
  console.error("网络错误:", err);
}
```

### 超时与中止（AbortController）

```js
const controller = new AbortController();
const timer = setTimeout(() => controller.abort(), 5000); // 5 秒超时

try {
  const res = await fetch("https://slow-api.example.com/data", {
    signal: controller.signal,
  });
  const data = await res.json();
} catch (err) {
  if (err.name === "AbortError") console.error("请求超时");
  else console.error(err);
} finally {
  clearTimeout(timer);
}
```

### `http` 模块 vs `fetch` 对比

| 维度 | `http.request` | `fetch` |
| --- | --- | --- |
| 风格 | 回调/事件流 | Promise / async-await |
| 是否需引入 | 需 `require("http")` | 内置全局，无需引入 |
| 响应体解析 | 手动拼接 `data` 事件 | `.json()` / `.text()` 一步到位 |
| 适用 | 底层控制、流式处理 | 日常 API 调用、现代推荐 |

## 静态文件服务器（fs + path + http 配合）

```js
const http = require("http");
const fs = require("fs/promises");
const path = require("path");

const server = http.createServer(async (req, res) => {
  try {
    // 把 / 映射到 index.html，其余路径拼到 public 目录
    const urlPath = req.url === "/" ? "/index.html" : req.url;
    const filePath = path.join(__dirname, "public", urlPath);
    const data = await fs.readFile(filePath);
    res.writeHead(200, { "Content-Type": "text/html" });
    res.end(data);
  } catch {
    res.writeHead(404, { "Content-Type": "text/plain" });
    res.end("Not Found");
  }
});

server.listen(3000);
```

## 小结

| 能力 | 关键 API |
| --- | --- |
| 创建服务器 | `http.createServer((req, res) => {...})` |
| 启动监听 | `server.listen(port, cb)` |
| 请求对象 | `req.url` / `req.method` / `req.headers` |
| 响应对象 | `res.writeHead()` / `res.end()` |
| 发起请求 | `http.request()` / `http.get()` / `fetch()` |
